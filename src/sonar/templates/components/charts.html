{# Server-rendered SVG charts (SPEC §12). Geometry comes from sonar.charts via the `charts` global, so these macros only draw. #} {% macro runway_bar(limit, balance, worst, best) %} {%- set width = 300 -%} {%- set g = charts.runway(limit, balance, worst, best, width) -%} {%- set summary = "Balance " ~ balance|eur ~ ", projected " ~ worst|eur ~ " to " ~ best|eur ~ " at payday, overdraft limit " ~ limit|eur -%} {%- endmacro %} {% macro month_columns(months) %} {%- set height = 80 -%} {%- set slot = 24 -%} {%- set heights = charts.columns(months|map(attribute="total_cents")|list, height) -%} {%- set peak = heights|max if heights else 0 -%} {%- endmacro %} {% macro category_bars(categories) %} {%- set width = 300 -%} {%- set row = 24 -%} {%- set widths = charts.bars(categories|map(attribute="median_cents")|list, width) -%} {%- endmacro %}
